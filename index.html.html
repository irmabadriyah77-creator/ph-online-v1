Panduan Setup Google Sheets & Apps Script

1 Struktur Sheet Google Sheets

Buat Spreadsheet baru di Google Drive, buat **3 Sheet** dengan nama persis berikut:

Sheet 1: Siswa

Kolom Baris 1: Nama | Password | Kelas

Sheet 2: Soal

Kolom Baris 1: ID Soal | Tipe Soal | Pertanyaan | Opsi A | Opsi B | Opsi C | Opsi D | Kunci Jawaban | Pembahasan

Sheet 3: Hasil

Kolom Baris 1: Nama Siswa | Tanggal/Waktu Submit | Skor Pilihan Ganda/Isian | Detail Jawaban Siswa | Status Koreksi Uraian

2 Kode Google Apps Script (Code.gs)

Buka menu **Ekstensi > Apps Script** pada Google Sheets Anda, hapus semua kode bawaan lalu tempel kode berikut:

function doGet(e) {
  var action = e.parameter.action;
  if (action === 'getQuestions') {
    return getQuestions();
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "API Penilaian Harian Aktif" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;

    if (action === 'login') {
      return handleLogin(contents.nama, contents.password);
    } else if (action === 'getQuestions') {
      return getQuestions();
    } else if (action === 'submitAnswers') {
      return submitAnswers(contents);
    }

    return responseJSON({ status: "error", message: "Action tidak dikenali" });
  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function handleLogin(nama, password) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Siswa");
  if (!sheet) return responseJSON({ status: "error", message: "Sheet 'Siswa' tidak ditemukan" });

  var data = sheet.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    var userNama = String(data[i][0]).trim();
    var userPass = String(data[i][1]).trim();
    var userKelas = String(data[i][2]).trim();

    if (userNama.toLowerCase() === String(nama).trim().toLowerCase() && userPass === String(password).trim()) {
      return responseJSON({
        status: "success",
        user: { nama: userNama, kelas: userKelas }
      });
    }
  }
  return responseJSON({ status: "error", message: "Nama atau Password tidak sesuai!" });
}

function getQuestions() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Soal");
  if (!sheet) return responseJSON({ status: "error", message: "Sheet 'Soal' tidak ditemukan" });

  var data = sheet.getDataRange().getValues();
  var questions = [];

  for (var i = 1; i < data.length; i++) {
    if (!data[i][0]) continue; // Skip jika ID kosong
    questions.push({
      id: data[i][0],
      tipe: data[i][1], // 'Pilihan Ganda', 'Isian', 'Uraian'
      pertanyaan: data[i][2],
      opsi: [data[i][3], data[i][4], data[i][5], data[i][6]].filter(Boolean),
      kunci: data[i][7],
      pembahasan: data[i][8] || "Tidak ada pembahasan khusus."
    });
  }

  return responseJSON({ status: "success", questions: questions });
}

function submitAnswers(payload) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetHasil = ss.getSheetByName("Hasil");
  if (!sheetHasil) {
    sheetHasil = ss.insertSheet("Hasil");
    sheetHasil.appendRow(["Nama Siswa", "Tanggal/Waktu Submit", "Skor Pilihan Ganda/Isian", "Detail Jawaban Siswa", "Status Koreksi Uraian"]);
  }

  var studentName = payload.nama;
  var answers = payload.jawaban; // Object: { soalId: userAns }
  var questions = payload.questions; // Questions array

  var totalAutograded = 0;
  var maxAutograded = 0;
  var hasUraian = false;

  questions.forEach(function(q) {
    var userAns = String(answers[q.id] || '').trim().toLowerCase();
    var keyAns = String(q.kunci || '').trim().toLowerCase();

    if (q.tipe === 'Pilihan Ganda' || q.tipe === 'Isian') {
      maxAutograded += 10;
      if (userAns === keyAns) {
        totalAutograded += 10;
      }
    } else if (q.tipe === 'Uraian') {
      hasUraian = true;
    }
  });

  // Skala nilai ke 100 untuk PG/Isian
  var finalScore = maxAutograded > 0 ? Math.round((totalAutograded / maxAutograded) * 100) : 0;
  var uraianStatus = hasUraian ? "Menunggu Koreksi Guru" : "Tidak Ada Uraian";

  var timestamp = new Date().toLocaleString("id-ID");
  var detailAnswersStr = JSON.stringify(answers);

  sheetHasil.appendRow([studentName, timestamp, finalScore, detailAnswersStr, uraianStatus]);

  return responseJSON({
    status: "success",
    score: finalScore,
    uraianStatus: uraianStatus
  });
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

3 Cara Deploy Menjadi Web App

  1. Klik tombol Deploy (Terapkan) > New deployment (Penerapan baru).
  2. Pilih jenis penerapan: Web App (Aplikasi Web).
  3. Isi Deskripsi (misal: "API PH v1").
  4. Atur Execute as (Jalankan sebagai): Me (Saya).
  5. Atur Who has access (Siapa yang memiliki akses): Anyone (Siapa saja).
  6. Klik Deploy, lalu berikan Izin Akses (Authorize Access).
  7. Salin Web App URL yang dihasilkan, lalu masukkan pada modal Pengaturan Backend di web ini.
